cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
project(PrecisionHold VERSION 0.10.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PrecisionHold requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
set_property(SOURCE resources/app.rc PROPERTY OBJECT_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/resources/precisionhold.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-normal-light.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-slowed-light.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-paused-light.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-normal-dark.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-slowed-dark.ico;${CMAKE_CURRENT_SOURCE_DIR}/resources/tray-paused-dark.ico")
add_executable(PrecisionHold WIN32 src/main.cpp src/pointer_speed.cpp src/settings_dialog.cpp src/tray_icon.cpp src/registry_store.cpp resources/app.rc)
target_include_directories(PrecisionHold PRIVATE resources)
target_link_libraries(PrecisionHold PRIVATE user32 advapi32 shell32 wtsapi32 comctl32 gdi32 gdiplus dwmapi uxtheme)
target_link_options(PrecisionHold PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/resources/app.manifest)
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
if(ipo_supported)
  set_property(TARGET PrecisionHold PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
install(TARGETS PrecisionHold RUNTIME DESTINATION .)
